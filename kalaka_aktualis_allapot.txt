function recordPersonalRoutine(userId, origin, dest) {
  if (!userId || !origin?.trim() || !dest?.trim()) return;
  try {
    const cleanO = origin.trim();
    const cleanD = dest.trim();
    if (cleanO.toLowerCase() === cleanD.toLowerCase()) return;
    const list = getPersonalRoutines(userId);
    const key = `${cleanO.toLowerCase()}->${cleanD.toLowerCase()}`;
    const existing = list.find((x) => x.key === key);
    if (existing) {
      existing.count = (existing.count || 1) + 1;
      existing.origin = cleanO;
      existing.dest = cleanD;
      existing.updatedAt = Date.now();
    } else {
      list.push({ key, origin: cleanO, dest: cleanD, count: 1, updatedAt: Date.now() });
    }
    list.sort((a, b) => (b.count - a.count) || (b.updatedAt - a.updatedAt));
    localStorage.setItem(`kalaka_routines_${userId}`, JSON.stringify(list.slice(0, 4)));
  } catch {}
}

function playSoftNotificationTone() {
  try {
    if (typeof navigator !== 'undefined' && navigator.vibrate) {
      navigator.vibrate([120, 60, 120]);
    }
    const AudioCtx = window.AudioContext || window.webkitAudioContext;
    if (!AudioCtx) return;
    const ctx = new AudioCtx();
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(660, ctx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(880, ctx.currentTime + 0.14);
    gain.gain.setValueAtTime(0.06, ctx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.22);
    osc.connect(gain);
    gain.connect(ctx.destination);
    osc.start();
    osc.stop(ctx.currentTime + 0.22);
  } catch {}
}

function triggerSystemNotification(title, body) {
  try {
    if (typeof window === 'undefined' || !('Notification' in window)) return;
    if (Notification.permission === 'granted' && document.visibilityState !== 'visible') {
      if (navigator.serviceWorker?.ready) {
        navigator.serviceWorker.ready
          .then((reg) => {
            reg.showNotification(title, {
              body,
              icon: '/icon.svg',
              badge: '/icon.svg',
              tag: 'kalaka-live-msg'
            });
          })
          .catch(() => {});
      } else {
        new Notification(title, { body, icon: '/icon.svg' });
      }
    }
  } catch {}
}

function getNextQuarterHourHHMM() {
  const d = new Date(Date.now() + 15 * 60000);
  const mins = Math.ceil(d.getMinutes() / 15) * 15;
  d.setMinutes(mins, 0, 0);
  return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`;
}

function getUpcomingRoundTimeSlots() {
  const fmt = (dt) =>
    `${String(dt.getHours()).padStart(2, '0')}:${String(dt.getMinutes()).padStart(2, '0')}`;

  const s1 = new Date(Date.now() + 5 * 60000);
  const rem = s1.getMinutes() % 15;
  s1.setMinutes(rem === 0 ? s1.getMinutes() : s1.getMinutes() + (15 - rem), 0, 0);

  const s2 = new Date(s1.getTime() + 15 * 60000);

  const s3 = new Date(s2.getTime());
  if (s2.getMinutes() === 0) {
    s3.setMinutes(30, 0, 0);
  } else {
    s3.setHours(s2.getHours() + 1, 0, 0, 0);
  }

  return [fmt(s1), fmt(s2), fmt(s3)];
}

function getOffsetTimeString(addMinutes = 10) {
  const d = new Date(Date.now() + addMinutes * 60000);
  const hh = String(d.getHours()).padStart(2, '0');
  const mm = String(d.getMinutes()).padStart(2, '0');
  return `${hh}:${mm}`;
}

function resolveAutoDayForTime(hhmm, isRecurring = false) {
  if (isRecurring) return 'Munkanapokon (H–P)';
  if (!hhmm || !hhmm.includes(':')) return 'Ma';
  const [hh, mm] = hhmm.split(':').map(Number);
  if (isNaN(hh) || isNaN(mm)) return 'Ma';
  const now = new Date();
  const target = new Date();
  target.setHours(hh, mm, 0, 0);
  return target.getTime() < now.getTime() - 60000 ? 'Holnap' : 'Ma';
}

function formatDayLabelFromIso(isoStr, isRec = false, schedLabel = null) {
  if (isRec) return schedLabel || 'Munkanapokon (H–P)';
  if (!isoStr) return 'Ma';
  const dt = new Date(isoStr);
  if (isNaN(dt.getTime())) return 'Ma';
  const now = new Date();
  const todayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
  const targetStart = new Date(dt.getFullYear(), dt.getMonth(), dt.getDate()).getTime();
  const diffDays = Math.round((targetStart - todayStart) / 86400000);
  if (diffDays <= 0) return 'Ma';
  if (diffDays === 1) return 'Holnap';
  return dt.toLocaleDateString('hu-HU', { month: 'short', day: 'numeric' });
}

export function stripAccents(str) {
  return (str || '')
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toLowerCase()
    .trim();
}

function createSessionToken() {
  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
    return crypto.randomUUID();
  }
  return `${Date.now()}-${Math.random().toString(36).slice(2, 11)}`;
}

function cleanHungarianAddress(rawText) {
  return (rawText || '')
    .replace(/,\s*Magyarország$/i, '')
    .replace(/,\s*Hungary$/i, '')
    .trim();
}

function expandStreetAbbreviations(addrStr) {
  return (addrStr || '')
    .replace(/\bu\.\s*/gi, 'utca ')
    .replace(/\bu\s+(?=\d)/gi, 'utca ')
    .replace(/\bkrt\.?\s*/gi, 'körút ')
    .replace(/\bstny\.?\s*/gi, 'sétány ')
    .trim();
}

function extractHouseNumber(rawQuery) {
  const m = (rawQuery || '').trim().match(/^(.*?)\s+(\d+[a-zA-Z/-]*)\.?$/);
  if (!m || m[1].trim().length < 2) {
    return { streetPart: (rawQuery || '').trim(), houseNum: '' };
  }
  return { streetPart: m[1].trim(), houseNum: `${m[2]}.` };
}

function formatPhotonFallback(feat, fallbackHouseNum = '') {
  const p = feat?.properties || {};
  const [lng, lat] = feat?.geometry?.coordinates || [];
  if (typeof lat !== 'number' || typeof lng !== 'number') return null;

  const city = p.city || p.town || p.village || p.municipality || '';
  const streetBase = p.street || p.name || '';
  const house = p.housenumber ? `${p.housenumber}.` : fallbackHouseNum;
  const withHouse =
    streetBase && house && !/\d/.test(streetBase)
      ? `${streetBase} ${house}`
      : streetBase || city;
  if (!withHouse) return null;

  const label =
    city && !stripAccents(withHouse).includes(stripAccents(city))
      ? `${city}, ${withHouse}`
      : withHouse;

  return {
    source: 'photon',
    label,
    subtitle: [p.postcode, city, p.district || p.suburb].filter(Boolean).join(' • ') || 'Magyarország',
    lat,
    lng
  };
}

function haversineMeters(lat1, lng1, lat2, lng2) {
  const R = 6371000;
  const toRad = (d) => (d * Math.PI) / 180;
  const dLat = toRad(lat2 - lat1);
  const dLng = toRad(lng2 - lng1);
  const a =
    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLng / 2) * Math.sin(dLng / 2);
  return 2 * R * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
}

    try {
      const cleanName = displayName.trim() || email.split('@')[0];

      if (mode === 'magic') {
        const { error } = await supabase.auth.signInWithOtp({
          email: email.trim(),
          options: {
            emailRedirectTo: window.location.origin,
            data: { full_name: cleanName, zone }
          }
        });
        if (error) throw error;
        setFeedback({ type: 'ok', text: 'Belépési link elküldve az e-mail címedre.' });
      } else if (mode === 'register') {
        if (!cleanName || cleanName.length < 2) {
          throw new Error('Add meg a keresztneved vagy beceneved.');
        }
        if (!adultConfirmed) {
          throw new Error('A regisztrációhoz pipáld be a 18+ megerősítést.');
        }

        const { data, error } = await supabase.auth.signUp({
          email: email.trim(),
          password,
          options: { data: { full_name: cleanName, zone } }
        });
        if (error) throw error;

        if (data?.user && inviteCodeInput.trim()) {

            .from('profiles')
            .select('id')
            .eq('invite_code', inviteCodeInput.trim().toUpperCase())
            .maybeSingle();

          if (inviter?.id) {
            await supabase
              .from('profiles')
              .update({ invited_by: inviter.id, display_name: cleanName, zone })
              .eq('id', data.user.id);
          }
        }

        if (data.session) {
          onClose();
        } else {
          setFeedback({ type: 'ok', text: 'Fiók létrehozva. Ellenőrizd az e-mail fiókodat.' });
        }
      } else {
        const { error } = await supabase.auth.signInWithPassword({
          email: email.trim(),
          password
        });
        if (error) throw error;
        onClose();
      }
    } catch (err) {
      setFeedback({ type: 'err', text: err.message });
    } finally {
      setLoading(false);
    }
  };

function isLivePresenceActive(pro) {
  if (!pro?.live_until || !pro?.live_lat || !pro?.live_lng) return false;
  return new Date(pro.live_until).getTime() > Date.now();
}

function formatRemainingLiveTime(liveUntil) {
  if (!liveUntil) return '';
  const diffMs = new Date(liveUntil).getTime() - Date.now();
  if (diffMs <= 0) return '';
  const mins = Math.ceil(diffMs / 60000);
  if (mins >= 60) {
    const h = Math.floor(mins / 60);
    const m = mins % 60;
    return m > 0 ? `${h} óra ${m} p` : `${h} óra`;
  }
  return `${mins} perc`;
}

function buildNavigationUrl(pro) {
  if (!pro) return null;
  const liveNow = isLivePresenceActive(pro);
  if (liveNow) {
    return `https://www.google.com/maps/dir/?api=1&destination=${pro.live_lat},${pro.live_lng}`;
  }
  if (pro.lat && pro.lng) {
    return `https://www.google.com/maps/dir/?api=1&destination=${pro.lat},${pro.lng}`;
  }
  if (pro.address_text && pro.address_text.trim()) {
    return `https://www.google.com/maps/dir/?api=1&destination=${encodeURIComponent(pro.address_text.trim())}`;
  }
  return null;
}

          .from('local_pros')

export function formatRemainingTime(expiresAtIso) {
  if (!expiresAtIso) return null;
  const diffMs = new Date(expiresAtIso).getTime() - Date.now();
  if (diffMs <= 0) return null;
  const totalMins = Math.ceil(diffMs / 60000);
  if (totalMins < 60) return `${totalMins} percig aktív`;
  const hours = Math.floor(totalMins / 60);
  if (hours < 24) return `${hours} óráig aktív`;
  const days = Math.ceil(hours / 24);
  return `${days} napig aktív`;
}

export function formatRelativeTime(isoString) {
  if (!isoString) return '';
  const diffSec = Math.max(1, Math.floor((Date.now() - new Date(isoString).getTime()) / 1000));
  if (diffSec < 60) return 'most';
  const diffMin = Math.floor(diffSec / 60);
  if (diffMin < 60) return `${diffMin} perce`;
  const diffHour = Math.floor(diffMin / 60);
  return `${diffHour} órája`;
}

export function resolveCoordsFromText(text) {
  if (!text) return null;
  const q = text.trim().toLowerCase();

  if (q.includes('hév') || q.includes('hev') || q.includes('állomás tér')) {
    return { lat: 47.6602, lng: 19.0738, label: 'Szentendre HÉV-végállomás' };
  }
  if (q.includes('fogoly')) {
    return { lat: 47.6918, lng: 19.0517, label: text };
  }
  if (q.includes('bükkös') || q.includes('bukkos')) {
    return { lat: 47.6701, lng: 19.0739, label: 'Bükkös-part' };
  }
  if (q.includes('papírgyár') || q.includes('papirgyar')) {
    return { lat: 47.6852, lng: 19.0732, label: '11-es út • Papírgyár' };
  }
  if (q.includes('pismány') || q.includes('pismany')) {
    return { lat: 47.6895, lng: 19.0555, label: 'Pismány' };
  }

  const exactOrPartial = LOCAL_PLACES.find(
    (p) => p.name.toLowerCase() === q || p.name.toLowerCase().includes(q)
  );
  if (exactOrPartial) {
    return { lat: exactOrPartial.lat, lng: exactOrPartial.lng, label: exactOrPartial.name };
  }

  const zone = SZENTENDRE_ZONES.find(
    (z) => z.name.toLowerCase().includes(q) || q.includes(z.name.toLowerCase())
  );
  if (zone) {
    return { lat: zone.lat, lng: zone.lng, label: zone.name };
  }

  if (q.includes('újpest') || q.includes('ujpest')) return { lat: 47.5605, lng: 19.0792, label: text };
  if (q.includes('batthyány') || q.includes('batthyany')) return { lat: 47.5065, lng: 19.0399, label: text };
  if (q.includes('árpád') || q.includes('göncz')) return { lat: 47.5325, lng: 19.0660, label: text };
  if (q.includes('hév') || q.includes('hev')) return { lat: 47.6602, lng: 19.0738, label: 'Szentendre HÉV-végállomás' };

  return null;
}

export async function searchAddressSuggestions(query) {
  const raw = (query || '').trim();
  if (raw.length < 2) return [];

  const qLower = raw.toLowerCase();
  const houseMatch = raw.match(/(?:^|\s)(\d+[a-zA-Z/.-]*)\s*$/);
  const typedHouseNumber = houseMatch ? houseMatch[1].replace(/\.*$/, '') : null;

  const localMatches = LOCAL_PLACES.filter(
    (p) =>
      p.name.toLowerCase().includes(qLower) ||
      p.subtitle.toLowerCase().includes(qLower)
  ).map((p) => {
    if (typedHouseNumber && !p.name.match(/\d/) && p.name.toLowerCase().includes('utca')) {
      return {
        ...p,
        name: `${p.name} ${typedHouseNumber}.`
      };
    }
    return p;
  });

  try {
    const cleanedForGeocoder = raw
      .replace(/^(felső-pismány|alsó-pismány|pismány|szarvashegy|izbég|sztaravoda|belváros)[,\s-]*/i, '')
      .trim();

    const searchTerm =
      cleanedForGeocoder.toLowerCase().includes('budapest') ||
      cleanedForGeocoder.toLowerCase().includes('újpest') ||
      cleanedForGeocoder.toLowerCase().includes('pomáz')
        ? cleanedForGeocoder
        : `${cleanedForGeocoder}, Szentendre`;

    const url = `https://nominatim.openstreetmap.org/search?format=jsonv2&addressdetails=1&countrycodes=hu&viewbox=18.95,47.76,19.20,47.48&bounded=1&limit=6&accept-language=hu&q=${encodeURIComponent(
      searchTerm
    )}`;

    const res = await fetch(url);
    const rows = await res.json();

    const remoteMatches = (rows || [])
      .map((item) => {
        const addr = item.address || {};
        const city =
          addr.city ||
          addr.town ||
          addr.village ||
          addr.suburb ||
          'Szentendre';
        const road = addr.road || addr.pedestrian || addr.residential || item.name || '';
        const house = addr.house_number
          ? `${addr.house_number}.`
          : typedHouseNumber && road
          ? `${typedHouseNumber}.`
          : '';

        const lat = parseFloat(item.lat);
        const lng = parseFloat(item.lon);
        if (!lat || !lng || (!road && !item.name)) return null;

        const mainTitle = road
          ? `${city === 'Szentendre' ? 'Szentendre, ' : city + ', '}${road}${house ? ' ' + house : ''}`
          : item.display_name.split(',').slice(0, 2).join(', ');

        return {
          name: mainTitle.trim(),
          subtitle: `${addr.postcode || '2000'} ${city}${addr.suburb ? ' • ' + addr.suburb : ''}`,
          lat,
          lng
        };
      })
      .filter(Boolean);

    const combined = [...localMatches];
    remoteMatches.forEach((rm) => {
      if (!combined.some((c) => c.name.toLowerCase() === rm.name.toLowerCase())) {
        combined.push(rm);
      }
    });

    return combined.slice(0, 6);
  } catch {
    return localMatches.slice(0, 6);
  }
}

}

function addMinutesToTime(timeStr, minsToAdd) {
  const [hh, mm] = (timeStr || '05:50').split(':').map(Number);
  const totalMins = (hh * 60 + mm + minsToAdd + 1440) % 1440;
  const nh = String(Math.floor(totalMins / 60)).padStart(2, '0');
  const nm = String(totalMins % 60).padStart(2, '0');
  return `${nh}:${nm}`;
}

function toMetersXY(lat, lng, refLat, refLng) {
  const latScale = 111132;
  const lngScale = 111132 * Math.cos((refLat * Math.PI) / 180);
  return {
    x: (lng - refLng) * lngScale,
    y: (lat - refLat) * latScale
  };
}

function projectPointToPolyline(pt, polyCoords) {
  if (!polyCoords || polyCoords.length < 2) return null;
  const refLat = pt.lat;
  const refLng = pt.lng;
  const P = toMetersXY(pt.lat, pt.lng, refLat, refLng);

  let bestDist = Infinity;
  let bestAlongMeters = 0;
  let cumMeters = 0;

  for (let i = 0; i < polyCoords.length - 1; i++) {
    const [lat1, lng1] = polyCoords[i];
    const [lat2, lng2] = polyCoords[i + 1];
    const A = toMetersXY(lat1, lng1, refLat, refLng);
    const B = toMetersXY(lat2, lng2, refLat, refLng);

    const dx = B.x - A.x;
    const dy = B.y - A.y;
    const segLenSq = dx * dx + dy * dy;
    const segLen = Math.sqrt(segLenSq);

    let t = 0;
    if (segLenSq > 0) {
      t = ((P.x - A.x) * dx + (P.y - A.y) * dy) / segLenSq;
      t = Math.max(0, Math.min(1, t));
    }

    const projX = A.x + t * dx;
    const projY = A.y + t * dy;
    const dist = Math.hypot(P.x - projX, P.y - projY);

    if (dist < bestDist) {
      bestDist = dist;
      bestAlongMeters = cumMeters + t * segLen;
    }

    cumMeters += segLen;
  }

  return {
    distanceMeters: bestDist,
    alongMeters: bestAlongMeters,
    totalMeters: cumMeters
  };
}

export async function fetchDrivingRoute(startCoord, endCoord) {
  if (!startCoord || !endCoord) return null;
  const key = `osrm_v2_${startCoord.lat.toFixed(4)},${startCoord.lng.toFixed(4)}->${endCoord.lat.toFixed(4)},${endCoord.lng.toFixed(4)}`;
  if (routeCache.has(key)) return routeCache.get(key);

  try {
    const cachedStr = typeof sessionStorage !== 'undefined' ? sessionStorage.getItem(key) : null;
    if (cachedStr) {
      const parsed = JSON.parse(cachedStr);
      if (parsed && Array.isArray(parsed.coords)) {
        routeCache.set(key, parsed);
        return parsed;
      }
    }
  } catch {}

  try {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), 3500);
    const url = `https://router.project-osrm.org/route/v1/driving/${startCoord.lng},${startCoord.lat};${endCoord.lng},${endCoord.lat}?overview=full&geometries=geojson`;
    const res = await fetch(url, { signal: controller.signal });
    clearTimeout(timer);
    const data = await res.json();
    if (data?.routes?.[0]) {
      const r = data.routes[0];
      const result = {
        coords: r.geometry.coordinates.map(([lng, lat]) => [lat, lng]),
        distanceMeters: r.distance,
        durationSeconds: r.duration
      };
      routeCache.set(key, result);
      try {
        if (typeof sessionStorage !== 'undefined') {
          sessionStorage.setItem(key, JSON.stringify(result));
        }
      } catch {}
      return result;
    }
  } catch {
    // Hálózati hiba vagy timeout esetén egyenes szakasz fallback
  }

  const fallback = {
    coords: [
      [startCoord.lat, startCoord.lng],
      [endCoord.lat, endCoord.lng]
    ],
    distanceMeters: 5000,
    durationSeconds: 600,
    isFallback: true
  };
  return fallback;
}

export async function computeStopsAlongRoute(ride) {
  if (!ride || ride.kind !== 'offer') return [];

  const startCoord =
    ride.origin_lat && ride.origin_lng
      ? { lat: ride.origin_lat, lng: ride.origin_lng }
      : resolveCoordsFromText(ride.origin_text);

  const endCoord =
    ride.dest_lat && ride.dest_lng
      ? { lat: ride.dest_lat, lng: ride.dest_lng }
      : resolveCoordsFromText(ride.dest_text);

  if (!startCoord || !endCoord) return [];

  const route = await fetchDrivingRoute(startCoord, endCoord);
  if (!route || !route.coords || route.coords.length < 2) return [];

  const maxCorridorMeters = route.isFallback ? 350 : 260;
  const candidates = [];

  for (const place of LOCAL_PLACES) {
    // Indulási és érkezési pont közvetlen környezetét (220 m) kihagyjuk
    const distFromStart = Math.hypot(
      (place.lat - startCoord.lat) * 111132,
      (place.lng - startCoord.lng) * 74830
    );
    const distFromEnd = Math.hypot(
      (place.lat - endCoord.lat) * 111132,
      (place.lng - endCoord.lng) * 74830
    );
    if (distFromStart < 220 || distFromEnd < 220) continue;

    const proj = projectPointToPolyline({ lat: place.lat, lng: place.lng }, route.coords);
    if (!proj || proj.totalMeters < 400) continue;

    const fraction = proj.alongMeters / proj.totalMeters;
    if (proj.distanceMeters <= maxCorridorMeters && fraction > 0.04 && fraction < 0.96) {
      candidates.push({
        name: place.name,
        lat: place.lat,
        lng: place.lng,
        alongMeters: proj.alongMeters,
        fraction
      });
    }
  }

  // Haladási irány szerinti sorbarendezés
  candidates.sort((a, b) => a.alongMeters - b.alongMeters);

  // Túl közeli (300 m-en belüli) megállók ritkítása
  const deduped = [];
  for (const c of candidates) {
    if (
      deduped.length === 0 ||
      c.alongMeters - deduped[deduped.length - 1].alongMeters >= 300
    ) {
      deduped.push(c);
    }
  }

  return deduped.slice(0, 6).map((c) => {
    const minsFromStart = Math.max(1, Math.round((c.fraction * route.durationSeconds) / 60));
    return {
      time: addMinutesToTime(ride.time_label, minsFromStart),
      name: c.name,
      lat: c.lat,
      lng: c.lng
    };
  });
}

export async function fetchStopsAlongRoute(routeWkt) {
  if (!supabase || !routeWkt) return SZENTENDRE_STOPS.slice(0, 6);
  const { data, error } = await supabase.rpc('get_stops_along_route', {
    p_route_wkt: routeWkt
  });
  if (error) {
    console.warn('PostGIS get_stops_along_route fallback:', error.message);
    return SZENTENDRE_STOPS.slice(0, 6);
  }
  return data;
}

// PostGIS RPC hívás: 200m-es folyosó-alapú útvonal párosítás
export async function matchRidesAlongCorridor(pickup, dropoff, radiusMeters = 200) {
  if (!supabase) return [];
  const { data, error } = await supabase.rpc('match_rides_along_corridor', {
    p_pickup_lng: pickup.lng,
    p_pickup_lat: pickup.lat,
    p_dropoff_lng: dropoff.lng,
    p_dropoff_lat: dropoff.lat,
    p_radius_m: radiusMeters
  });
  if (error) throw error;
  return data || [];
}
